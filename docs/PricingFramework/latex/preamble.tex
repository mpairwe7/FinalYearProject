  \ifx\relax#1\relax\else

  \fi}

\newcommand{\PartHead}[3]{%
  \clearpage
  \phantomsection
  \addcontentsline{toc}{section}{\texorpdfstring{#1\quad #2}{#1 #2}}%
  \markboth{\MakeUppercase{#1\ \textperiodcentered\ #2}}{}%
  {\color{ink}\hrule height 1.6pt}%
  \vspace{3mm}%
  \gutter{#3}%
  {\datafont\fontsize{7}{9}\selectfont\textcolor{slate}{\MakeUppercase{#1}}\par}
  \vspace{1.4mm}%
  {\displayfont\fontsize{19}{21}\selectfont\textcolor{ink}{#2}\par}
  \vspace{3.5mm}\par\nointerlineskip}
